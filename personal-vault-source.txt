# Personal Vault PWA

A personal password + banking vault built around the agreed Apple Ã— Fintech Ã— Security visual direction.

## 1. What you get

- React + Vite PWA
- Supabase Auth + PostgreSQL
- Client-side Argon2id key derivation
- AES-GCM encryption using Web Crypto
- IndexedDB encrypted cache
- Password Management
  - Social Media
  - Web Password
  - Project Password
- Bank / ATM Card swipe stack
- Secure Notes
- Light / Dark / System
- Realistic / Abstract banking card switch
- Compact dashboard
- Password reveal with eye icon and auto-hide
- Encrypted JSON export
- Local CSV import

## 2. Install

Node.js 20+ is recommended.

```powershell
cd .\personal-vault
npm install
```

Copy environment template:

```powershell
Copy-Item .env.example .env.local
```

Edit `.env.local`:

```env
VITE_SUPABASE_URL=https://YOUR_PROJECT.supabase.co
VITE_SUPABASE_PUBLISHABLE_KEY=sb_publishable_...
```

Only the publishable key belongs in browser code. Do not put a Supabase secret key or service-role key in `.env.local` for Vite frontend use.


RLS is enabled and all policies are scoped to `auth.uid()`.

## 4. Auth settings

For the easiest first test, email/password auth must be enabled in Supabase Auth.

If email confirmation is enabled, create the account and confirm the email before signing in. For a local-only first test, you may disable email confirmation in the Auth provider settings, then turn it back on for a real deployment.

## 5. Start

```powershell
npm run dev
```

Open the localhost URL shown by Vite.

## 6. First vault setup

1. Create a Supabase account in the app.
2. Create your master password when prompted.
3. Unlock the vault.
4. Add passwords or bank cards.

The master password is never stored. The database receives encrypted `payload` objects only.

## 7. Important security note

This is a serious-secret-management project. Treat V1 as a personal prototype, not as a substitute for an audited password manager. The browser crypto path is designed around standard primitives, but the application itself has not undergone a security audit.

Do not store actual PINs, CVVs, OTP secrets, seed phrases, recovery codes, or other high-risk secrets in normal fields until a threat model and audited workflow are added.

function parseCsv(text){
  const rows=[]; let row=[], cell='', quoted=false
  for(let i=0;i<text.length;i+=1){
    const ch=text[i], next=text[i+1]
    if(ch==='"' && quoted && next==='"'){ cell+='"'; i+=1; continue }
    if(ch==='"'){ quoted=!quoted; continue }
    if(ch===',' && !quoted){ row.push(cell.trim()); cell=''; continue }
    if((ch==='\n' || ch==='\r') && !quoted){ if(ch==='\r' && next==='\n') i+=1; row.push(cell.trim()); cell=''; if(row.some(Boolean)) rows.push(row); row=[]; continue }
    cell+=ch
  }
  if(cell.length || row.length){ row.push(cell.trim()); if(row.some(Boolean)) rows.push(row) }
  if(!rows.length)return []
  const headers=rows.shift().map((h)=>h.toLowerCase().replace(/^\ufeff/,''));
  return rows.map((values)=>Object.fromEntries(headers.map((h,i)=>[h,values[i]||''])))
}

function formatNumber(value, mode) {
  const digits = (value || '').replace(/\D/g, '')
  if (mode === 'hidden') return `â€¢â€¢â€¢â€¢ â€¢â€¢â€¢â€¢ â€¢â€¢${digits.slice(-2).padStart(2, 'â€¢')}`

  for (let i = 0; i < digits.length; i += 4) parts.push(digits.slice(i, i + 4))

}

export default function Modal({
  title,
  eyebrow,
  children,
  onClose,
  wide = false
}) {
  useEffect(() => {
    const handleKeyDown = (event) => {
      if (event.key === 'Escape') {
        onClose?.()
      }
    }

    document.addEventListener('keydown', handleKeyDown)

    return () => {
      document.removeEventListener('keydown', handleKeyDown)
    }
  }, [onClose])

  return (
    <div
      className="sheet-backdrop"
      onMouseDown={onClose}
      role="presentation"
    >
      <section
        className={wide ? 'modal wide' : 'modal'}
        role="dialog"
        aria-modal="true"
        aria-labelledby="modal-title"
        onMouseDown={(event) => event.stopPropagation()}
      >
        <div className="sheet-header">
          <div>
            {eyebrow && (
              <div className="eyebrow">{eyebrow}</div>
            )}

            <h3 id="modal-title">{title}</h3>
          </div>

          <button
            type="button"
            className="icon-button"
            onClick={onClose}
            aria-label="Close"
            title="Close"
          >
            <X size={19} />
          </button>
        </div>

      </section>
    </div>
  )
}

import { useEffect, useState } from 'react'
import {
  faviconCandidates,
  siteInitial,
  siteTint
} from '../lib/favicon'

/**
 * Shows the logo for a saved URL.
 *
 * Tries the site's own icon files in order. If none load, falls back to a
 * coloured initial derived from the domain. No third-party favicon service
 * is contacted, so the list of saved sites never leaves the device.
 */
export default function SiteLogo({
  url,
  name,
  size = 18,
  radius = 6,
  className = 'site-logo'
}) {
  const candidates = faviconCandidates(url)
  const [index, setIndex] = useState(0)
  const [failed, setFailed] = useState(false)

  useEffect(() => {
    setIndex(0)
    setFailed(false)
  }, [url])

  const hasCandidate =
    !failed && index < candidates.length

  const box = {
    width: `${size}px`,
    height: `${size}px`,
    borderRadius: `${radius}px`
  }

  if (!hasCandidate) {
    return (
      <span
        className={`${className} site-logo-fallback`}
        style={{
          ...box,
          background: siteTint(url || name),
          fontSize: `${Math.max(8, Math.round(size * 0.46))}px`
        }}
        aria-hidden="true"
      >
        {siteInitial(name, url)}
      </span>
    )
  }

  return (
    <img
      className={className}
      style={box}
      src={candidates[index]}
      alt=""
      loading="lazy"
      decoding="async"
      referrerPolicy="no-referrer"
      onError={() => {
        if (index + 1 < candidates.length) {
          setIndex(index + 1)
        } else {
          setFailed(true)
        }
      }}
    />
  )
}

import { useEffect, useState } from 'react'
import { Fingerprint, Lock, ShieldCheck } from './Icons'

  hasBiometricEnrollment,

export default function UnlockScreen({
  profile,
  user,
  onUnlocked
}) {
  const [masterPassword, setMasterPassword] = useState('')
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState('')
  const [biometricBusy, setBiometricBusy] = useState(false)

  const userId = user?.id
  const biometricReady =
    isBiometricSupported() &&
    hasBiometricEnrollment(userId)

  useEffect(() => {
    setMasterPassword('')
    setError('')
  }, [profile?.id])

  async function unlock(event) {
    event.preventDefault()

    setBusy(true)
    setError('')

    try {
      const key = deriveVaultKey(
        masterPassword,
        profile.vault_salt
      )

      const valid = await verifyMasterKey(profile, key)

      if (!valid) {
        throw new Error('Master password is incorrect.')
      }

      onUnlocked(key)
    } catch (err) {
      setError(err.message || 'Unable to unlock vault.')
    } finally {
      setBusy(false)
    }
  }

  async function unlockWithDevice() {
    setBiometricBusy(true)
    setError('')

    try {
      const key = await unlockWithBiometric(userId)
      onUnlocked(key)
    } catch (err) {
      setError(err.message || 'Biometric unlock failed.')
    } finally {
      setBiometricBusy(false)
    }
  }

  return (
    <main className="unlock-screen">
      <div className="unlock-orb">
        <ShieldCheck size={28} />
      </div>

      <div className="unlock-copy">
        <div className="eyebrow">Secure vault</div>

        <p>
          Unlock your private space. Your master password never
          leaves this device.

            autoFocus
            value={masterPassword}
            onChange={(e) =>
              setMasterPassword(e.target.value)
            }

            placeholder="Enter master password"

// Device biometric unlock using WebAuthn + the PRF extension.
//
// How it works:
//  1. The user enables biometrics while the vault is unlocked, so the raw
//     vault key is in memory.
//  2. We create a platform credential (Touch ID / Windows Hello / Android
//     biometrics) that supports the PRF extension.
//  3. The authenticator derives a stable 32-byte secret (the PRF output) from
//     a random per-enrollment salt. That secret never leaves the device.
//  4. We encrypt the raw vault key with that PRF secret and store only the
//     ciphertext locally. The stored record is useless without the
//     authenticator, so the vault key is never written to disk in the clear.
//
// Consequence: biometric unlock is per device. A new device needs the master
// password once, then it can enroll its own biometrics.

const STORE_PREFIX = 'vault.biometric.'

function bufferToBase64Url(buffer) {
  const bytes = new Uint8Array(buffer)
  let binary = ''

  for (let i = 0; i < bytes.length; i += 1) {
    binary += String.fromCharCode(bytes[i])
  }

  return btoa(binary)
    .replace(/\+/g, '-')
    .replace(/\//g, '_')
    .replace(/=+$/, '')
}

function base64UrlToBytes(value) {
  const normalized = String(value)
    .replace(/-/g, '+')
    .replace(/_/g, '/')

  const padding =
    normalized.length % 4 === 0
      ? ''
      : '='.repeat(4 - (normalized.length % 4))

  const binary = atob(normalized + padding)
  const bytes = new Uint8Array(binary.length)

  for (let i = 0; i < binary.length; i += 1) {
    bytes[i] = binary.charCodeAt(i)
  }

  return bytes
}

function importAesKey(raw) {
  return crypto.subtle.importKey(
    'raw',
    raw,
    { name: 'AES-GCM' },
    false,
    ['encrypt', 'decrypt']
  )
}

export function isBiometricSupported() {
  return (
    typeof window !== 'undefined' &&
    typeof window.PublicKeyCredential !== 'undefined' &&
    typeof navigator?.credentials?.create === 'function' &&
    typeof navigator?.credentials?.get === 'function'
  )
}

export async function isPlatformAuthenticatorAvailable() {
  if (!isBiometricSupported()) return false

  try {
    return await window.PublicKeyCredential.isUserVerifyingPlatformAuthenticatorAvailable()
  } catch {
    return false
  }
}

export function getBiometricRecord(userId) {
  if (!userId) return null

  try {
    const raw = localStorage.getItem(
      STORE_PREFIX + userId
    )

    return raw ? JSON.parse(raw) : null
  } catch {
    return null
  }
}

export function hasBiometricEnrollment(userId) {
  return Boolean(getBiometricRecord(userId))
}

export function removeBiometricRecord(userId) {
  if (!userId) return

  try {
    localStorage.removeItem(STORE_PREFIX + userId)
  } catch {
    /* storage unavailable */
  }
}

async function evaluatePrf(credentialId, prfSalt) {
  let assertion

  try {
    assertion = await navigator.credentials.get({
      publicKey: {
        challenge: crypto.getRandomValues(
          new Uint8Array(32)
        ),
        rpId: window.location.hostname,
        allowCredentials: [
          {
            type: 'public-key',
            id: base64UrlToBytes(credentialId)
          }
        ],
        userVerification: 'required',
        timeout: 60000,
        extensions: {
          prf: { eval: { first: prfSalt } }
        }
      }
    })
  } catch (error) {
    if (error?.name === 'NotAllowedError') {
      throw new Error('Biometric verification was cancelled.')
    }

    throw error
  }

  if (!assertion) {
    throw new Error('Biometric verification was cancelled.')
  }

  const results = assertion.getClientExtensionResults?.()
  const first = results?.prf?.results?.first

  if (!first) {
    throw new Error(
      'This device cannot derive a biometric key. Use your master password instead.'
    )
  }

  return new Uint8Array(first)
}

export async function enrollBiometric(
  userId,
  { email, displayName } = {},
  vaultKey
) {
  if (!isBiometricSupported()) {
    throw new Error(
      'This browser does not support biometric unlock.'
    )
  }

  if (!vaultKey) {
    throw new Error('Unlock your vault before enabling biometrics.')
  }

  let credential

  try {
    credential = await navigator.credentials.create({
      publicKey: {
        challenge: crypto.getRandomValues(
          new Uint8Array(32)
        ),
        rp: {
          name: 'Personal Vault',
          id: window.location.hostname
        },
        user: {
          id: new TextEncoder().encode(userId),
          name: email || userId,
          displayName: displayName || email || 'Vault user'
        },
        pubKeyCredParams: [
          { type: 'public-key', alg: -7 },
          { type: 'public-key', alg: -257 }
        ],
        authenticatorSelection: {
          authenticatorAttachment: 'platform',
          userVerification: 'required',
          residentKey: 'preferred'
        },
        timeout: 60000,
        attestation: 'none',
        extensions: { prf: {} }
      }
    })
  } catch (error) {
    if (error?.name === 'NotAllowedError') {
      throw new Error('Biometric setup was cancelled.')
    }

    throw error
  }

  if (!credential) {
    throw new Error('Biometric setup was cancelled.')
  }

  const credentialId = bufferToBase64Url(
    credential.rawId
  )

  // The PRF output is only produced during an assertion, so we immediately
  // evaluate it once to wrap the vault key.
  const prfSalt = crypto.getRandomValues(
    new Uint8Array(32)
  )

  const prfOutput = await evaluatePrf(
    credentialId,
    prfSalt
  )

  const wrappingKey = await importAesKey(prfOutput)
  const iv = crypto.getRandomValues(
    new Uint8Array(12)
  )

  const cipher = await crypto.subtle.encrypt(
    { name: 'AES-GCM', iv },
    wrappingKey,
    vaultKey
  )

  const record = {
    version: 1,
    credentialId,
    prfSalt: bufferToBase64Url(prfSalt),
    iv: bufferToBase64Url(iv),
    wrapped: bufferToBase64Url(cipher),
    createdAt: new Date().toISOString()
  }

  try {
    localStorage.setItem(
      STORE_PREFIX + userId,
      JSON.stringify(record)
    )
  } catch {
    throw new Error('Unable to store biometric data on this device.')
  }

  return record
}

export async function unlockWithBiometric(userId) {
  const record = getBiometricRecord(userId)

  if (!record) {
    throw new Error(
      'No biometric unlock is set up on this device.'
    )
  }

  const prfOutput = await evaluatePrf(
    record.credentialId,
    base64UrlToBytes(record.prfSalt)
  )

  const wrappingKey = await importAesKey(prfOutput)

  try {
    const plain = await crypto.subtle.decrypt(
      { name: 'AES-GCM', iv: base64UrlToBytes(record.iv) },
      wrappingKey,
      base64UrlToBytes(record.wrapped)
    )

    return new Uint8Array(plain)
  } catch {
    throw new Error(
      'Biometric unlock failed on this device. Sign in with your master password and enable it again.'
    )
  }
}

import { argon2id } from '@noble/hashes/argon2.js'
import { utf8ToBytes, randomBytes, bytesToHex } from '@noble/hashes/utils.js'

const textEncoder = new TextEncoder()
const textDecoder = new TextDecoder()

// Browser-friendly starting point. Tune after profiling target mobile devices.
const ARGON_OPTIONS = {
  t: 3,
  m: 64 * 1024,
  p: 1,
  dkLen: 32,
  maxmem: 128 * 1024 * 1024
}

function bytesToBase64(bytes) {
  let binary = ''
  const chunk = 0x8000
  for (let i = 0; i < bytes.length; i += chunk) {
    binary += String.fromCharCode(...bytes.subarray(i, i + chunk))
  }
  return btoa(binary)
}

function base64ToBytes(value) {
  const binary = atob(value)
  const bytes = new Uint8Array(binary.length)
  for (let i = 0; i < binary.length; i += 1) bytes[i] = binary.charCodeAt(i)
  return bytes
}

export function generateSalt() {
  return bytesToBase64(randomBytes(16))
}

export function generateId() {
  return crypto.randomUUID()
}

export function deriveVaultKey(masterPassword, saltBase64) {
  const salt = base64ToBytes(saltBase64)
  return argon2id(utf8ToBytes(masterPassword), salt, ARGON_OPTIONS)
}

async function importAesKey(rawKey) {
  return crypto.subtle.importKey('raw', rawKey, { name: 'AES-GCM' }, false, ['encrypt', 'decrypt'])
}

export async function encryptJson(value, rawKey) {
  const iv = crypto.getRandomValues(new Uint8Array(12))
  const key = await importAesKey(rawKey)
  const plaintext = textEncoder.encode(JSON.stringify(value))
  const cipher = await crypto.subtle.encrypt({ name: 'AES-GCM', iv }, key, plaintext)
  return {
    v: 1,
    iv: bytesToBase64(iv),
    ct: bytesToBase64(new Uint8Array(cipher))
  }
}

export async function decryptJson(envelope, rawKey) {
  const key = await importAesKey(rawKey)
  const plaintext = await crypto.subtle.decrypt(
    { name: 'AES-GCM', iv: base64ToBytes(envelope.iv) },
    key,
    base64ToBytes(envelope.ct)
  )
  return JSON.parse(textDecoder.decode(plaintext))
}

export function stableJson(value) {
  return JSON.stringify(value, Object.keys(value).sort())
}

export function keyFingerprint(rawKey) {
  return bytesToHex(rawKey.slice(0, 6))
}

// Privacy-respecting site logos.
//
// We never send the list of saved sites to a third-party favicon service.
// Instead we try the site's own icon files first and fall back to a
// deterministic coloured initial when nothing loads.

export function siteDomain(input) {
  if (!input) return ''

  let value = String(input).trim()

  if (!value) return ''

  if (!/^[a-z][a-z0-9+.-]*:\/\//i.test(value)) {
    value = `https://${value}`
  }

  try {
    const host = new URL(value).hostname.toLowerCase()
    return host.replace(/^www\./, '')
  } catch {
    return ''
  }
}

export function faviconCandidates(input) {
  const domain = siteDomain(input)

  if (!domain || !domain.includes('.')) return []

  return [
    `https://${domain}/favicon.ico`,
    `https://${domain}/favicon.png`,
    `https://${domain}/apple-touch-icon.png`
  ]
}

export function siteInitial(name, input) {
  const source =
    String(name || '').trim() || siteDomain(input)

  const match = source.match(/[a-z0-9]/i)

  return match ? match[0].toUpperCase() : '?'
}

export function siteTint(input) {
  const domain = siteDomain(input) || String(input || '')

  let hash = 0

  for (let i = 0; i < domain.length; i += 1) {
    hash = (hash * 31 + domain.charCodeAt(i)) % 360
  }

  return `hsl(${hash} 58% 45%)`
}

const DB_NAME = 'personal-vault'
const DB_VERSION = 1
const STORE = 'vault_items'

function openDb() {
  return new Promise((resolve, reject) => {
    const request = indexedDB.open(DB_NAME, DB_VERSION)
    request.onupgradeneeded = () => {
      const db = request.result
      if (!db.objectStoreNames.contains(STORE)) {
        const store = db.createObjectStore(STORE, { keyPath: 'id' })
        store.createIndex('user_id', 'user_id', { unique: false })
        store.createIndex('updated_at', 'updated_at', { unique: false })
      }
    }
    request.onsuccess = () => resolve(request.result)
    request.onerror = () => reject(request.error)
  })
}

export async function idbPut(item) {
  const db = await openDb()
  await new Promise((resolve, reject) => {
    const tx = db.transaction(STORE, 'readwrite')
    tx.objectStore(STORE).put(item)
    tx.oncomplete = resolve
    tx.onerror = () => reject(tx.error)
  })
  db.close()
}

export async function idbBulkPut(items) {
  const db = await openDb()
  await new Promise((resolve, reject) => {
    const tx = db.transaction(STORE, 'readwrite')
    const store = tx.objectStore(STORE)
    items.forEach((item) => store.put(item))
    tx.oncomplete = resolve
    tx.onerror = () => reject(tx.error)
  })
  db.close()
}

export async function idbGetUserItems(userId) {
  const db = await openDb()
  const result = await new Promise((resolve, reject) => {
    const tx = db.transaction(STORE, 'readonly')
    const index = tx.objectStore(STORE).index('user_id')
    const req = index.getAll(IDBKeyRange.only(userId))
    req.onsuccess = () => resolve(req.result || [])
    req.onerror = () => reject(req.error)
  })
  db.close()
  return result
}

export async function idbClearUser(userId) {
  const items = await idbGetUserItems(userId)
  if (!items.length) return
  const db = await openDb()
  await new Promise((resolve, reject) => {
    const tx = db.transaction(STORE, 'readwrite')
    const store = tx.objectStore(STORE)
    items.forEach((item) => store.delete(item.id))
    tx.oncomplete = resolve
    tx.onerror = () => reject(tx.error)
  })
  db.close()
}

export async function ensureProfile(user, displayName) {
  const { data, error } = await supabase
    .from('profiles')
    .select('id, display_name, avatar_url, vault_salt, vault_verifier, theme, card_style')
    .eq('id', user.id)
    .maybeSingle()
  if (error) throw error
  if (data) return data

  const { data: inserted, error: insertError } = await supabase
    .from('profiles')
    .insert({ id: user.id, display_name: displayName || user.email?.split('@')[0] || 'You' })
    .select('id, display_name, avatar_url, vault_salt, vault_verifier, theme, card_style')
    .single()
  if (insertError) throw insertError
  return inserted
}

    .eq('user_id', userId)
    .order('updated_at', { ascending: false })

  const rows = data || []
  await idbBulkPut(rows)
  return rows

export async function uploadAvatar(userId, file) {
  if (!userId) {
    throw new Error('User ID is required')
  }

  if (!file) {
    throw new Error('Avatar file is required')
  }

  const allowedTypes = [
    'image/jpeg',
    'image/png',
    'image/webp'
  ]

  if (!allowedTypes.includes(file.type)) {
    throw new Error('Avatar must be JPG, PNG, or WebP')
  }

  if (file.size > 5 * 1024 * 1024) {
    throw new Error('Avatar must be smaller than 5 MB')
  }

  const filePath = `${userId}/avatar.webp`

  const { error: uploadError } = await supabase.storage
    .from('avatars')
    .upload(filePath, file, {
      cacheControl: '3600',
      upsert: true,
      contentType: 'image/webp'
    })

  if (uploadError) {
    throw uploadError
  }

  const { data: publicUrlData } = supabase.storage
    .from('avatars')
    .getPublicUrl(filePath)

  const avatarUrl =
    publicUrlData?.publicUrl || filePath

  const { error: profileError } = await supabase
    .from('profiles')
    .update({
      avatar_url: avatarUrl,
      updated_at: new Date().toISOString()
    })
    .eq('id', userId)

  if (profileError) {
    throw profileError
  }

  return avatarUrl
}

import { useEffect, useState } from 'react'
import Modal from '../components/Modal'
import SiteLogo from '../components/SiteLogo'
import {
  Copy,
  Eye,
  EyeOff,
  Pencil,
  Trash2,
  Star,
  Plus,
  X
} from '../components/Icons'

export default function ItemModal({
  item,
  onClose,
  onDelete,
  onSave,
  onToggleFavorite
}) {
  const [editing, setEditing] = useState(false)
  const [form, setForm] = useState(normalizeItem(item))
  const [visibleSecrets, setVisibleSecrets] = useState({})

  useEffect(() => {
    setForm(normalizeItem(item))
    setEditing(false)
    setVisibleSecrets({})
  }, [item])

  const update = (key, value) => {
    setForm((current) => ({
      ...current,
      [key]: value
    }))
  }

  const toggleSecret = (key) => {
    setVisibleSecrets((current) => ({
      ...current,
      [key]: !current[key]
    }))
  }

  const save = () => {
    if (!String(form.name || '').trim()) {
      return
    }

    const cleanedCustomFields = (form.custom_fields || [])
      .filter(
        (field) =>
          String(field.label || '').trim() ||
          String(field.value || '').trim()
      )
      .map(({ label, value, secret }) => ({
        label: String(label || '').trim(),
        value: value || '',
        secret: Boolean(secret)
      }))

    onSave({
      ...form,
      name: String(form.name).trim(),
      custom_fields: cleanedCustomFields
    })

    setForm((current) => ({
      ...current,
      custom_fields: cleanedCustomFields
    }))

    setEditing(false)
    setVisibleSecrets({})
  }

  const cancelEdit = () => {
    setForm(normalizeItem(item))
    setEditing(false)
    setVisibleSecrets({})
  }

  const addCustomField = () => {
    setForm((current) => ({
      ...current,
      custom_fields: [
        ...(current.custom_fields || []),
        {
          id: crypto.randomUUID(),
          label: '',
          value: '',
          secret: false
        }
      ]
    }))
  }

  const updateCustomField = (id, key, value) => {
    setForm((current) => ({
      ...current,
      custom_fields: (current.custom_fields || []).map((field) =>
        field.id === id
          ? {
              ...field,
              [key]: value
            }
          : field
      )
    }))
  }

  const removeCustomField = (id) => {
    setForm((current) => ({
      ...current,
      custom_fields: (current.custom_fields || []).filter(
        (field) => field.id !== id
      )
    }))
  }

  const isFavorite = Boolean(item.favorite)

  return (
    <Modal
      title={
        item.item_type === 'bank_card'
          ? 'Card details'
          : item.item_type === 'secure_note'
            ? 'Secure note'
            : item.name
      }
      eyebrow={formatType(item.item_type)}
      wide
      onClose={onClose}
    >
      <div className="detail-header">
        <div className="detail-identity">
          {form.url && (
            <SiteLogo
              url={form.url}
              name={form.name}
              size={34}
              radius={10}
            />
          )}

          <div>
            <span className="detail-name">
              {form.name || 'Untitled'}
            </span>

            <span className="detail-category">
              {form.categoryLabel || formatType(item.item_type)}
            </span>
          </div>
        </div>

        <div className="button-row">
          <button
            type="button"
            className={
              isFavorite
                ? 'favorite-button selected'
                : 'favorite-button'
            }
            onClick={() => onToggleFavorite(item.id)}
          >
            <Star
              size={15}
              fill={isFavorite ? 'currentColor' : 'none'}
            />
            Favorite
          </button>

          {!editing ? (
            <button
              type="button"
              className="secondary-button compact"
              onClick={() => setEditing(true)}
            >
              <Pencil size={15} />
              Edit
            </button>
          ) : (
            <button
              type="button"
              className="secondary-button compact"
              onClick={cancelEdit}
            >
              Cancel
            </button>
          )}

          <button
            type="button"
            className="danger-ghost compact"
            onClick={() => onDelete(item.id)}
            title="Delete"
            aria-label="Delete"
          >
            <Trash2 size={15} />
          </button>
        </div>
      </div>

      {!editing ? (
        <ReadOnly
          item={form}
          visibleSecrets={visibleSecrets}
          toggleSecret={toggleSecret}
        />
      ) : (
        <EditForm
          item={form}
          update={update}
          visibleSecrets={visibleSecrets}
          toggleSecret={toggleSecret}
          addCustomField={addCustomField}
          updateCustomField={updateCustomField}
          removeCustomField={removeCustomField}
        />
      )}

      {editing && (
        <div className="modal-actions">
          <button
            type="button"

            onClick={cancelEdit}
          >

function normalizeItem(item) {
  return {
    ...item,

    name: item.name || '',
    username: item.username || '',
    password: item.password || '',
    url: item.url || '',
    notes: item.notes || '',
    environment: item.environment || '',
    service: item.service || '',

    bank_name: item.bank_name || '',
    account_owner: item.account_owner || '',
    card_number: item.card_number || '',
    account_number: item.account_number || '',
    registered_phone: item.registered_phone || '',
    card_brand: item.card_brand || '',
    expiry: item.expiry || '',
    color: item.color || '',

    note: item.note || '',

    custom_fields: (item.custom_fields || []).map((field) => ({
      id: field.id || crypto.randomUUID(),
      label: field.label || '',
      value: field.value || '',
      secret: Boolean(field.secret)
    }))
  }
}

function formatType(type) {
  if (type === 'bank_card') {
    return 'Bank / ATM Card'
  }

  if (type === 'secure_note') {
    return 'Secure Note'
  }

  if (type === 'password') {
    return 'Password'
  }

  return String(type || '').replaceAll('_', ' ')
}

create extension if not exists pgcrypto;

create table if not exists public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  display_name text not null default 'You',
  avatar_url text,
  vault_salt text,
  vault_verifier jsonb,
  theme text not null default 'system' check (theme in ('light','dark','system')),
  card_style text not null default 'realistic' check (card_style in ('realistic','abstract')),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists public.vault_items (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  item_type text not null check (item_type in ('password','bank_card','secure_note')),
  category text not null,
  payload jsonb not null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create index if not exists idx_vault_items_user_updated on public.vault_items(user_id, updated_at desc);
create index if not exists idx_vault_items_user_type on public.vault_items(user_id, item_type);

create or replace function public.set_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

drop trigger if exists profiles_set_updated_at on public.profiles;
create trigger profiles_set_updated_at
before update on public.profiles
for each row execute procedure public.set_updated_at();

drop trigger if exists vault_items_set_updated_at on public.vault_items;
create trigger vault_items_set_updated_at
before update on public.vault_items
for each row execute procedure public.set_updated_at();

alter table public.profiles enable row level security;
alter table public.vault_items enable row level security;

drop policy if exists profiles_select_own on public.profiles;
create policy profiles_select_own on public.profiles
for select to authenticated
using (id = auth.uid());

drop policy if exists profiles_insert_own on public.profiles;
create policy profiles_insert_own on public.profiles
for insert to authenticated
with check (id = auth.uid());

drop policy if exists profiles_update_own on public.profiles;
create policy profiles_update_own on public.profiles
for update to authenticated
using (id = auth.uid())
with check (id = auth.uid());

drop policy if exists vault_items_select_own on public.vault_items;
create policy vault_items_select_own on public.vault_items
for select to authenticated
using (user_id = auth.uid());

drop policy if exists vault_items_insert_own on public.vault_items;
create policy vault_items_insert_own on public.vault_items
for insert to authenticated
with check (user_id = auth.uid());

drop policy if exists vault_items_update_own on public.vault_items;
create policy vault_items_update_own on public.vault_items
for update to authenticated
using (user_id = auth.uid())
with check (user_id = auth.uid());

drop policy if exists vault_items_delete_own on public.vault_items;
create policy vault_items_delete_own on public.vault_items
for delete to authenticated
using (user_id = auth.uid());

create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer set search_path = public
as $$
begin
  insert into public.profiles (id, display_name)
  values (new.id, coalesce(new.raw_user_meta_data ->> 'display_name', split_part(coalesce(new.email, 'You'), '@', 1)))
  on conflict (id) do nothing;
  return new;
end;
$$;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
after insert on auth.users
for each row execute procedure public.handle_new_user();

-- Supabase Data API privileges for browser client.

grant select, insert, update, delete on public.vault_items to authenticated;

-- Storage bucket for user avatars.
--
-- Run this after 001_schema.sql. The bucket is public for reads (avatars are
-- not secrets) but every write is scoped to the owner's folder, which is
-- <user_id>/..., so one user can never overwrite another user's file.

insert into storage.buckets (id, name, public)
values ('avatars', 'avatars', true)
on conflict (id) do update set public = excluded.public;

drop policy if exists avatars_read_all on storage.objects;
create policy avatars_read_all on storage.objects
for select
to public
using (bucket_id = 'avatars');

drop policy if exists avatars_insert_own on storage.objects;
create policy avatars_insert_own on storage.objects
for insert
to authenticated
with check (
  bucket_id = 'avatars'
  and (storage.foldername(name))[1] = auth.uid()::text
);

drop policy if exists avatars_update_own on storage.objects;
create policy avatars_update_own on storage.objects
for update
to authenticated
using (
  bucket_id = 'avatars'
  and (storage.foldername(name))[1] = auth.uid()::text
)
with check (
  bucket_id = 'avatars'
  and (storage.foldername(name))[1] = auth.uid()::text
);

drop policy if exists avatars_delete_own on storage.objects;
create policy avatars_delete_own on storage.objects
for delete
to authenticated
using (
  bucket_id = 'avatars'
  and (storage.foldername(name))[1] = auth.uid()::text
);
